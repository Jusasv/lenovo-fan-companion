cmake_minimum_required(VERSION 3.20)
project(LenovoFanCompanion VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(CTest)
add_executable(diagnostic-tests diagnostic_tests.cpp)
if(MSVC)
  target_compile_options(diagnostic-tests PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(diagnostic-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME diagnostic-rules COMMAND diagnostic-tests)
add_executable(manual-tests manual_tests.cpp)
if(MSVC)
  target_compile_options(manual-tests PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(manual-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME manual-sequence COMMAND manual-tests)

if(WIN32)
  add_executable(LenovoFanManual WIN32 manual.cpp manual.manifest)
  target_compile_definitions(LenovoFanManual PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(LenovoFanManual PRIVATE advapi32 user32 gdi32)
  add_executable(LenovoFanCompanion WIN32 diagnostics.cpp app.manifest)
  target_compile_definitions(LenovoFanCompanion PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(LenovoFanCompanion PRIVATE setupapi cfgmgr32 advapi32 ole32 oleaut32 wbemuuid comdlg32 shell32 user32 gdi32)
  if(MSVC)
    target_compile_options(LenovoFanManual PRIVATE /W4 /WX /utf-8)
    target_link_options(LenovoFanManual PRIVATE "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/manual.manifest" "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
    target_compile_options(LenovoFanCompanion PRIVATE /W4 /WX /utf-8)
    target_link_options(LenovoFanCompanion PRIVATE "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
    set_property(TARGET LenovoFanCompanion LenovoFanManual diagnostic-tests manual-tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endif()
